#include <stdio.h>

int main() {
    int n, i;
    float sum = 0.0, score;

    printf("Enter number of subjects: ");
    scanf("%d", &n);

    for (i = 1; i <= n; i++) {
        printf("Enter score %d: ", i);
        scanf("%f", &score);
        sum += score;
    }

    if (n > 0) {
        printf("Average score: %.2f\n", sum / n);
    }
    return 0;
}
